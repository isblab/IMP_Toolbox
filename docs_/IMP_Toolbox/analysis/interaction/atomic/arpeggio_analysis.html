 34def get_interactions(
 35    ri_df: pd.DataFrame | None = None,
 36    contacts_df: pd.DataFrame | None = None,
 37    ari_df: pd.DataFrame | None = None,
 38    contact_level: str = "residue",
 39) -> pd.DataFrame:
 40    """ Combine interactions from contacts, ri, and ari dataframes.
 41
 42    ## Arguments:
 43
 44    - **ri_df (pd.DataFrame | None, optional):**:<br />
 45        Ring-ring interactions dataframe.
 46
 47    - **contacts_df (pd.DataFrame | None, optional):**:<br />
 48        Atomic interactions dataframe.
 49
 50    - **ari_df (pd.DataFrame | None, optional):**:<br />
 51        Atom-ring interactions dataframe.
 52
 53    - **contact_level (str, optional):**:<br />
 54        Contact level: "atom" or "residue". Determines the columns included in the output dataframe and how interactions are categorized. Must be one of "atom" or "residue".
 55
 56    ## Returns:
 57
 58    - **pd.DataFrame**:<br />
 59        DataFrame containing combined interactions. The columns included depend on the specified contact level. Interactions from the input dataframes are combined and categorized based on the specified contact level.
 60    """
 61
 62    assert contact_level in INTERACTION_DF_COLS, (
 63        f"contact_level must be one of {list(INTERACTION_DF_COLS.keys())}."
 64    )
 65
 66    combined_df = pd.DataFrame(columns=INTERACTION_DF_COLS[contact_level])
 67
 68    if contacts_df is not None:
 69
 70        contacts_df = contacts_df.reset_index(drop=True)
 71        combined_df = pd.concat(
 72            [
 73                combined_df,
 74                contacts_df.filter(INTERACTION_DF_COLS[contact_level])
 75            ],
 76            ignore_index=True
 77        )
 78
 79        for idx, row in contacts_df.iterrows():
 80            detected_interactions = [
 81                interaction
 82                for interaction in CHOSEN_CONTACT_TYPES + CHOSEN_CLASHES
 83                if row[interaction] == '1'
 84            ]
 85            if len(detected_interactions) == 0:
 86                continue
 87
 88            detected_interactions = ",".join(detected_interactions)
 89            combined_df.iloc[idx, -1] = detected_interactions
 90
 91        combined_df = combined_df.dropna(
 92            subset=["interaction_type"]
 93        ).reset_index(drop=True)
 94
 95    if ri_df is not None:
 96
 97        ri_df = ri_df.reset_index(drop=True)
 98        if contact_level == "residue":
 99            ri_df["interaction_type"] = "PI-PI_STACKING"
100
101        combined_df = pd.concat(
102            [
103                combined_df,
104                ri_df.filter(INTERACTION_DF_COLS[contact_level])
105            ],
106            ignore_index=True
107        )
108
109    if ari_df is not None:
110
111        ari_df = ari_df.reset_index(drop=True)
112        combined_df = pd.concat(
113            [
114                combined_df,
115                ari_df.filter(INTERACTION_DF_COLS[contact_level])
116            ],
117            ignore_index=True
118        )
119
120    combined_df = combined_df.drop_duplicates().reset_index(drop=True)
121
122    return combined_df

151def run_arpeggio_docker(
152    docker_base_command: Template,
153    processed_struct_path: str,
154    docker_result_dir: str,
155    result_head: str,
156    result_metadata: dict,
157    dry_run: bool=False,
158    overwrite: bool=False,
159) -> None:
160    """ Run Arpeggio analysis inside a Docker container.
161
162    Note: Assumes that at least a subset of the selections are in the
163    structure. If no part of the selection is in the structure, it will
164    raise an error.
165
166    ## Arguments:
167
168    - **docker_base_command (Template)**:<br />
169        Template for the Docker command to run Arpeggio.
170
171    - **processed_struct_path (str)**:<br />
172        Path to the processed structure file on the host machine. This file will be copied to the Docker mount path for analysis.
173
174    - **docker_result_dir (str)**:<br />
175        Path on the host machine where Docker results will be stored. The processed structure file will be copied here, and Arpeggio results will be saved in this directory.
176
177    - **result_head (str)**:<br />
178        Identifier for the result set. This will be used to name the processed structure file in the Docker mount path and to organize Arpeggio results.
179
180    - **result_metadata (dict)**:<br />
181        Metadata for the result set, including selections.
182
183    - **dry_run (bool, optional):**:<br />
184        If True, only print the Docker command without executing it.
185
186    - **overwrite (bool, optional):**:<br />
187        If True, overwrite existing results. If False and results already exist for the given result_head, the function will skip execution and print a warning.
188    """
189
190    path_to_mount = os.path.join(docker_result_dir, result_head)
191    if os.path.exists(path_to_mount) and overwrite is False:
192        warnings.warn(
193            f"""Arpeggio results already exist for {result_head} at \
194            {path_to_mount}. Skipping.
195            To overwrite, set overwrite=True."""
196        )
197        return  # already processed
198
199    processed_struct_path = os.path.abspath(processed_struct_path)
200    if not os.path.isfile(processed_struct_path):
201        raise Exception(
202            f"Processed structure file not found for \
203            {result_head}: {processed_struct_path}."
204        )
205
206    os.makedirs(path_to_mount, exist_ok=True)
207    # copy processed structure to mount path
208    target_path = os.path.join(path_to_mount, f"{result_head}.pdb")
209    os.system(f"cp {processed_struct_path} {target_path}")
210
211    docker_command = docker_base_command.substitute(
212        path_to_mount=path_to_mount,
213        container_path=os.path.dirname(target_path),
214        input_pdb_path=target_path,
215    )
216
217    arpeggio_sels = result_metadata.get("selections", [""])
218
219    selection_strs = []
220    for arpeggio_sel in arpeggio_sels:
221
222        if len(arpeggio_sel) == 0:
223            continue
224
225        _, chain, res, _ = arpeggio_sel.split("/")
226
227        if len(res) == 0:
228            selection_strs.append(f"/{chain}/")
229        else:
230            res_list = get_res_range_from_key(res)
231            selection_strs.extend([
232                f"/{chain}/{res}/" for res in res_list
233            ])
234
235    docker_command = docker_command.strip() + f" -s {' '.join(selection_strs)}"
236
237    print(f"Running Docker command for {result_head}:\n{docker_command}\n")
238    if dry_run is False:
239        os.system(docker_command)
240    print("=" * 100 + "\n")

264    def __init__(self, result_dir: str):
265        self.result_dir = result_dir
266        self.result_head = os.path.basename(result_dir)
267        self.contacts_path = os.path.join(result_dir, f"{self.result_head}.contacts")
268        self.rings_path = os.path.join(result_dir, f"{self.result_head}.rings")
269        self.ri_path = os.path.join(result_dir, f"{self.result_head}.ri")
270        self.ari_path = os.path.join(result_dir, f"{self.result_head}.ari")

272    def parse_contacts(
273        self,
274        split_atom_col: bool=True
275    ) -> pd.DataFrame | None:
276        """ Parses the .contacts file generated by Arpeggio and returns a dataframe.
277
278        > [!NOTE]
279        > The atoms in the .contacts file are represented as:
280        >    `chain/res_num/atom`
281        >
282        > This function splits these into three separate columns depending on the
283        > value of `split_atom_col`.
284        >
285        > If True, the function will split the `atom_1` and `atom_2` columns into
286        > `chain_1`, `res_1`, `atom_1` and `chain_2`, `res_2`, `atom_2`.
287        >
288        > If False, the original `atom_1` and `atom_2` columns will be retained
289        > without splitting.
290
291        ## Arguments:
292
293        - **split_atom_col (bool, optional):**:<br />
294            Whether to split the atom columns into chain, res, and atom.
295
296        ## Returns:
297
298        - **pd.DataFrame | None**:<br />
299            DataFrame containing the parsed contact information. If the .contacts
300            file is empty, returns None.
301        """
302
303        with open(self.contacts_path, 'r') as f:
304            lines = f.readlines()
305
306        if len(lines) == 0:
307            return None
308
309        df = pd.DataFrame(columns=CONTACTS_FIELDS)
310
311        for line in lines:
312            parts = line.split("\t")
313            row_to_add = {k: v for k, v in zip(CONTACTS_FIELDS, parts)}
314            df = pd.concat([df, pd.DataFrame([row_to_add])], ignore_index=True)
315
316        if not split_atom_col:
317            return df
318
319        df[["chain_1", "res_1", "atom_1"]] = df["atom_1"].str.split("/", expand=True)
320        df[["chain_2", "res_2", "atom_2"]] = df["atom_2"].str.split("/", expand=True)
321
322        front_cols = [
323            "chain_1", "res_1", "atom_1",
324            "chain_2", "res_2", "atom_2"
325        ]
326        df = df[front_cols + [col for col in df.columns if col not in front_cols]]
327
328        return df

330    def parse_ri(
331        self,
332        split_residue_col: bool=True,
333        add_marker_id: bool=True
334    ) -> pd.DataFrame | None:
335        """ Parse ring-ring interactions from .ri file.
336
337        ## Arguments:
338
339        - **split_residue_col (bool, optional):**:<br />
340            Whether to split the `ring1_residue` and `ring2_residue` columns into
341            `chain_1`, `res_1` and `chain_2`, `res_2` respectively.
342
343        - **add_marker_id (bool, optional):**:<br />
344            Whether to add marker IDs for the interacting rings.
345            If True, the function will parse the .rings file to create a mapping
346            from ring_id to marker_id and add `marker_id_1` and `marker_id_2` columns
347            to the output dataframe corresponding to `ring1_id` and `ring2_id` respectively.
348
349        ## Returns:
350
351        - **pd.DataFrame | None**:<br />
352            DataFrame containing the parsed ring interaction information. If the .ri
353            file is empty, returns None.
354        """
355
356        with open(self.ri_path, 'r') as f:
357            lines = f.readlines()
358
359        if len(lines) == 0:
360            return None
361
362        if add_marker_id:
363            rings_df = self.parse_rings(
364                add_marker_id=True,
365                split_residue_col=False
366            )
367            r_to_m = {
368                row["ring_id"]: row["marker_id"]
369                for _, row in rings_df.iterrows()
370            }
371
372        df = pd.DataFrame(columns=RI_FIELDS)
373
374        for line in lines:
375            parts = line.split("\t")
376            row_to_add = {k: v for k, v in zip(RI_FIELDS, parts)}
377            row_to_add["ring1_centroid"] = ",".join(
378                [str(x) for x in eval(row_to_add["ring1_centroid"])]
379            )
380            row_to_add["ring2_centroid"] = ",".join(
381                [str(x) for x in eval(row_to_add["ring2_centroid"])]
382            )
383            if add_marker_id:
384                row_to_add["marker_id_1"] = r_to_m.get(row_to_add["ring1_id"], "")
385                row_to_add["marker_id_2"] = r_to_m.get(row_to_add["ring2_id"], "")
386
387            df = pd.concat([df, pd.DataFrame([row_to_add])], ignore_index=True)
388
389        if not split_residue_col:
390            return df
391
392        df[["chain_1", "res_1", "_"]] = df["ring1_residue"].str.split("/", expand=True)
393        df[["chain_2", "res_2", "_"]] = df["ring2_residue"].str.split("/", expand=True)
394
395        del df["ring1_residue"]
396        del df["ring2_residue"]
397        del df["_"]
398
399        front_cols = [
400            "chain_1", "res_1",
401            "chain_2", "res_2"
402        ]
403        df = df[front_cols + [col for col in df.columns if col not in front_cols]]
404
405        return df

407    def parse_rings(
408        self,
409        add_marker_id: bool=True,
410        split_residue_col: bool=True
411    ) -> pd.DataFrame | None:
412        """ Parses the .rings file.
413
414        ## Arguments:
415
416        - **add_marker_id (bool, optional):**:<br />
417            Whether to add marker IDs to the output dataframe.
418
419        - **split_residue_col (bool, optional):**:<br />
420            Whether to split the `ring_residue` column into `chain`, `res`, and `_` columns.
421
422        ## Returns:
423
424        - **pd.DataFrame | None**:<br />
425            DataFrame containing the parsed ring information. If the .rings file is empty, returns None.
426        """
427
428        with open(self.rings_path, 'r') as f:
429            lines = f.readlines()
430
431        if len(lines) == 0:
432            return None
433
434        df = pd.DataFrame(columns=RING_FIELDS)
435
436        for idx, line in enumerate(lines):
437            parts = line.split("\t")
438            row_to_add = {k: v for k, v in zip(RING_FIELDS, parts)}
439            row_to_add["ring_centroid"] = ",".join(
440                [str(x) for x in eval(row_to_add["ring_centroid"])]
441            )
442
443            if add_marker_id:
444                row_to_add["marker_id"] = f"{idx+1}"
445
446            df = pd.concat([df, pd.DataFrame([row_to_add])], ignore_index=True)
447
448        if not split_residue_col:
449            return df
450
451        df[["chain", "res", "_"]] = df["ring_residue"].str.split("/", expand=True)
452
453        del df["ring_residue"]
454        del df["_"]
455
456        front_cols = ["chain", "res", "marker_id"]
457        df = df[front_cols + [col for col in df.columns if col not in front_cols]]
458
459        return df

461    def parse_ari(
462        self,
463        split_atom_col: bool=True,
464        split_residue_col: bool=True,
465        add_marker_id: bool=True,
466    ) -> pd.DataFrame | None:
467        """ Parse the atom-ring interactions from .ari file.
468
469        ## Arguments:
470
471        - **split_atom_col (bool, optional):**:<br />
472            Whether to split ``atom`` column into `chain_1`, `res_1`, and `atom_1` columns.
473
474        - **split_residue_col (bool, optional):**:<br />
475            Whether to split `residue` column into `chain_2` and `res_2` columns.
476
477        - **add_marker_id (bool, optional):**:<br />
478            Whether to add marker IDs for the interacting rings.
479
480        ## Returns:
481
482        - **pd.DataFrame | None**:<br />
483            DataFrame containing the parsed atom-ring interaction information. If the .ari
484            file is empty, returns None.
485        """
486
487        with open(self.ari_path, 'r') as f:
488            lines = f.readlines()
489
490        if len(lines) == 0:
491            return None
492
493        if add_marker_id:
494            rings_df = self.parse_rings(
495                add_marker_id=True,
496                split_residue_col=False
497            )
498            r_to_m = {
499                row["ring_id"]: row["marker_id"]
500                for _, row in rings_df.iterrows()
501            }
502
503        df = pd.DataFrame(columns=ARI_FIELDS)
504
505        for _, line in enumerate(lines):
506            parts = line.split("\t")
507            row_to_add = {k: v for k, v in zip(ARI_FIELDS, parts)}
508            row_to_add["ring_centroid"] = ",".join(
509                [str(x) for x in eval(row_to_add["ring_centroid"])]
510            )
511
512            row_to_add["interaction_type"] = ",".join(
513                [itype.strip() for itype in eval(row_to_add["interaction_type"])]
514            )
515
516            if add_marker_id:
517                row_to_add["marker_id"] = r_to_m.get(row_to_add["ring_id"], "")
518
519            df = pd.concat([df, pd.DataFrame([row_to_add])], ignore_index=True)
520
521        if not split_atom_col and not split_residue_col:
522            return df
523
524        df[["chain_2", "res_2", "_"]] = df["residue"].str.split("/", expand=True)
525        del df["residue"]
526        del df["_"]
527        front_cols = ["chain_2", "res_2", "ring_id"]
528        df = df[front_cols + [col for col in df.columns if col not in front_cols]]
529
530        df[["chain_1", "res_1", "atom_1"]] = df["atom"].str.split("/", expand=True)
531        del df["atom"]
532        front_cols = ["chain_1", "res_1", "atom_1"]
533        df = df[front_cols + [col for col in df.columns if col not in front_cols]]
534
535        return df

571    @staticmethod
572    def add_contacts(
573        contacts_df: pd.DataFrame | None = None,
574        save_path: str | None = None,
575        add_selections: bool = True,
576    ) -> list[str]:
577        """ Generate ChimeraX commands to add pseudobonds for atomic contacts.
578
579        ## Arguments:
580
581        - **contacts_df (pd.DataFrame | None, optional):**:<br />
582            DataFrame containing inter-atomic contact information.
583
584        - **save_path (str | None, optional):**:<br />
585            Path to save the commands. If None, returns the commands as a list.
586
587        - **add_selections (bool, optional):**:<br />
588            Whether to add selection commands for the interacting residues.
589
590        ## Returns:
591
592        - **list**:<br />
593            List of ChimeraX commands.
594        """
595
596        if contacts_df is None or contacts_df.empty:
597            warnings.warn("No contacts found.")
598            return []
599
600        command_template = Template(PBOND_COMMAND)
601        transparency_template = Template(TRANSPARENCY_COMMAND)
602        # model_id = 1
603        commands = []
604        sub_model_ids = {}
605        sub_model_id = 1
606
607        for _, row in contacts_df.iterrows():
608
609            atom1_spec = f"{MODEL_ID}/{row['chain_1']}:{row['res_1']}@{row['atom_1']}"
610            atom2_spec = f"{MODEL_ID}/{row['chain_2']}:{row['res_2']}@{row['atom_2']}"
611
612            if add_selections:
613                sel_cmd = f"sel add #{atom1_spec.split("@")[0]}#{atom2_spec.split('@')[0]}"
614                if sel_cmd not in commands:
615                    commands.append(sel_cmd)
616
617            detected_interactions = [
618                interaction for interaction in CHOSEN_CONTACT_TYPES + CHOSEN_CLASHES
619                if row[interaction] == '1'
620            ]
621
622            for interaction in detected_interactions:
623
624                if interaction not in sub_model_ids:
625                    sub_model_ids[interaction] = sub_model_id
626                    sub_model_id += 1
627
628                command = command_template.substitute(
629                    spec1=atom1_spec,
630                    spec2=atom2_spec,
631                    bond_color=PBOND_ATTRIBUTES.get(
632                        interaction, PBOND_ATTRIBUTES["default"]
633                        ).get("color", "yellow"),
634                    bond_radius=PBOND_ATTRIBUTES.get(
635                        interaction, PBOND_ATTRIBUTES["default"]
636                        ).get("radius", 0.1),
637                    bond_name=interaction,
638                    bond_dashes=PBOND_ATTRIBUTES.get(
639                        interaction, PBOND_ATTRIBUTES["default"]
640                        ).get("dashes", 6),
641                )
642                commands.append(command)
643
644                command = transparency_template.substitute(
