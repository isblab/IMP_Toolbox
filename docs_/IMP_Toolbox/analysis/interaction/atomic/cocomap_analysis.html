 48def run_cocomap_docker(
 49    processed_struct_path: str,
 50    docker_result_dir: str,
 51    result_head: str,
 52    result_metadata: dict,
 53    docker_base_command: Template = Template(DOCKER_BASE_COMMAND),
 54    dry_run: bool = False,
 55    overwrite: bool = False,
 56):
 57    """ Run COCOMAPS 2 analysis using its docker image.
 58
 59    ## Arguments:
 60
 61    - **processed_struct_path (str)**:<br />
 62        Path to processed structure file (PDB) to analyze.
 63
 64    - **docker_result_dir (str)**:<br />
 65        Directory where COCOMAP results will be stored.
 66
 67    - **result_head (str)**:<br />
 68        Unique identifier for the result (used for naming output directory).
 69
 70    - **result_metadata (dict)**:<br />
 71        Metadata dictionary containing information about the result, including:
 72
 73    - **docker_base_command (Template, optional):**:<br />
 74        Docker command template to run COCOMAP. Defaults to DOCKER_BASE_COMMAND.
 75
 76    - **dry_run (bool, optional):**:<br />
 77        If True, only print the docker command without executing it. Defaults to False.
 78
 79    - **overwrite (bool, optional):**:<br />
 80        If True, overwrite existing results. If False and results already exist
 81        for the given result_head, the function will skip execution and print a warning.
 82    """
 83
 84    cocomap_output_dir = os.path.join(docker_result_dir, result_head)
 85    cocomap_output_dir = os.path.abspath(cocomap_output_dir)
 86
 87    if os.path.exists(cocomap_output_dir) and overwrite is False:
 88        warnings.warn(
 89            f"""COCOMAP results already exist for {result_head} at
 90            {cocomap_output_dir}. Skipping.
 91            To overwrite, set overwrite=True."""
 92        )
 93        return  # already processed
 94
 95    processed_struct_path = os.path.abspath(processed_struct_path)
 96    if not os.path.isfile(processed_struct_path):
 97        raise Exception(
 98            f"Processed structure file not found for \
 99            {result_head}: {processed_struct_path}."
100        )
101
102    struct_type = os.path.splitext(processed_struct_path)[1].lower()
103
104    os.makedirs(cocomap_output_dir, exist_ok=True)
105
106    # copy processed structure to the result directory
107    target_pdb_path = os.path.join(cocomap_output_dir, f"{result_head}{struct_type}")
108    os.system(f"cp {processed_struct_path} {target_pdb_path}")
109
110    docker_command = docker_base_command.substitute(
111        path_to_mount=cocomap_output_dir,
112        config_path=os.path.join(cocomap_output_dir, "config.json")
113    )
114
115    # write config file required by COCOMAP to the result directory
116    config_params = TEMPLATE_CONFIG.copy()
117    config_params.update(
118        {
119            "chains_set_1": result_metadata["chains_set_1"],
120            "chains_set_2": result_metadata["chains_set_2"],
121            "ranges_1": result_metadata["ranges_1"],
122            "ranges_2": result_metadata["ranges_2"],
123            "pdb_file": target_pdb_path,
124        }
125    )
126    config_path = os.path.join(cocomap_output_dir, "config.json")
127    write_json(config_path, config_params)
128
129    print(f"Running Docker command for {result_head}:\n{docker_command}\n")
130    if not dry_run:
131        os.system(docker_command)
132        os.system(f"sudo chown -R $USER:$USER {cocomap_output_dir}")
133    print("=" * 100 + "\n")

135def postprocess_cocomap_results(
136    result_metadata: dict,
137    docker_results_dir: str,
138    result_head: str,
139):
140    """ Rename COCOMAP CSVs columns by protein names & explode by interaction types.
141
142    ## Arguments:
143
144    - **result_metadata (dict)**:<br />
145        Metadata dictionary containing information about the result, including
146        'proteins' key with list of protein names for chain1 & 2.
147
148    - **docker_results_dir (str)**:<br />
149        Directory where COCOMAP results are stored.
150
151    - **result_head (str)**:<br />
152        Unique identifier for the result (used for naming output directory).
153    """
154
155    cocomap_output_dir = os.path.join(docker_results_dir, result_head)
156    cocomap_output_dir = os.path.abspath(cocomap_output_dir)
157
158    proteins = result_metadata.get("proteins", [])
159    assert len(proteins) in [0, 2], "proteins must 0 or 2"
160    prot_1, prot_2 = proteins[0], proteins[1]
161    new_cols = {
162        "Chain 1": prot_1,
163        "Chain 2": prot_2,
164    }
165
166    csv_files = [
167        f for f in os.listdir(cocomap_output_dir) if f.endswith(".csv")
168    ]
169    # Rename chain columns to protein names
170    for csv_file in csv_files:
171        csv_path = os.path.join(cocomap_output_dir, csv_file)
172        df = pd.read_csv(csv_path, index_col=0, encoding='utf-8')
173        df = df.rename(columns=new_cols)
174        df.to_csv(csv_path)
175
176    final_csvs = [
177        f for f in os.listdir(cocomap_output_dir)
178        if f.endswith("final_file.csv")
179    ]
180
181    # Explode final CSVs by interaction types to keep one interaction per row
182    for final_csv in final_csvs:
183        final_csv_path = os.path.join(cocomap_output_dir, final_csv)
184        df = pd.read_csv(final_csv_path, index_col=0, encoding='utf-8')
185        df["Type of Interactions"] = df["Type of Interactions"].str.split("; ")
186        df_exploded = df.explode("Type of Interactions")
187        df_exploded = df_exploded[
188            df_exploded["Type of Interactions"].str.strip() != ""
189        ]
190        df_exploded = df_exploded.drop_duplicates()
191        df_exploded.to_csv(final_csv_path, index=False)

193def get_mutation_annotated_df(
194    cocomap_df: pd.DataFrame,
195    mutation1_df: pd.DataFrame,
196    mutation2_df: pd.DataFrame,
197    af_missense_dict1: dict,
198    af_missense_dict2: dict,
199    p_name1: str,
200    p_name2: str,
201) -> pd.DataFrame:
202    """ Add mutation annotations to COCOMAP dataframe.
203
204    ## Arguments:
205
206    - **cocomap_df (pd.DataFrame)**:<br />
207        DataFrame containing COCOMAP interaction data.
208
209    - **mutation1_df (pd.DataFrame)**:<br />
210        DataFrame containing mutation information for protein 1 in the output
211        format of clinvar_mutations.py
212
213    - **mutation2_df (pd.DataFrame)**:<br />
214        DataFrame containing mutation information for protein 2 in the output
215        format of clinvar_mutations.py
216
217    - **af_missense_dict1 (dict)**:<br />
218        Dictionary containing AlphaFold missense mutation information for protein 1,
219        in the output format of af_missense_mutations.py
220
221    - **af_missense_dict2 (dict)**:<br />
222        Dictionary containing AlphaFold missense mutation information for protein 2,
223        in the output format of af_missense_mutations.py
224
225    - **p_name1 (str)**:<br />
226        Name of protein 1 (used as key to access af_missense_dict1).
227
228    - **p_name2 (str)**:<br />
229        Name of protein 2 (used as key to access af_missense_dict2).
230
231    ## Returns:
232
233    - **pd.DataFrame**:<br />
234        COCOMAP DataFrame with added columns for mutated residues and AlphaFold missense scores.
235    """
236
237    df_res_1 = list(cocomap_df["Res. Number 1"].unique())
238    df_res_2 = list(cocomap_df["Res. Number 2"].unique())
239
240    mutations1 = mutation1_df["Mutation"].tolist()
241    mutations2 = mutation2_df["Mutation"].tolist()
242
243    clinvar1_patho = mutation1_df["ClinVar clinical significance"].tolist()
244    clinvar2_patho = mutation2_df["ClinVar clinical significance"].tolist()
245
246    mutated_res_1 = {}
247    mutated_res_2 = {}
248    afm_score_res_1 = {}
249    afm_score_res_2 = {}
250    afm_patho_mutations_1 = {}
251    afm_patho_mutations_2 = {}
252
253    for res1 in df_res_1:
254        mutated_res_1[res1] = [
255            mut for idx, mut in enumerate(mutations1)
256            if str(res1) == split_missense_mutation(mut)[1]
257            # and af_missense1_patho[idx] == "Likely pathogenic"
258            and clinvar1_patho[idx] in ["Likely pathogenic", "Pathogenic", "Conflicting classifications of pathogenicity", "Pathogenic/Likely pathogenic"]
259        ]
260        afm_score_res_1[res1] = [
261            af_missense_dict1[p_name1][mut]["patho_score"]
262            for idx, mut in enumerate(af_missense_dict1[p_name1])
263            if str(res1) == split_missense_mutation(mut)[1]
264        ]
265        afm_patho_mutations_1_ = [
266            AMINO_ACID_MAP[af_missense_dict1[p_name1][mut]["mut_aa"]]
267            for idx, mut in enumerate(af_missense_dict1[p_name1])
268            if str(res1) == split_missense_mutation(mut)[1]
269            and af_missense_dict1[p_name1][mut]["v_pathogenicity"] == "Likely pathogenic"
270        ]
271        afm_patho_mutations_1[res1] = f"({", ".join(afm_patho_mutations_1_)})"
272
273    for res2 in df_res_2:
274        mutated_res_2[res2] = [
275            mut for idx, mut in enumerate(mutations2)
276            if str(res2) == split_missense_mutation(mut)[1]
277            # and af_missense2_patho[idx] == "Likely pathogenic"
278            and clinvar2_patho[idx] in ["Likely pathogenic", "Pathogenic", "Conflicting classifications of pathogenicity", "Pathogenic/Likely pathogenic"]
279        ]
280        afm_score_res_2[res2] = [
281            af_missense_dict2[p_name2][mut]["patho_score"]
282            for idx, mut in enumerate(af_missense_dict2[p_name2])
283            if str(res2) == split_missense_mutation(mut)[1]
284        ]
285        afm_patho_mutations_2_ = [
286            AMINO_ACID_MAP[af_missense_dict2[p_name2][mut]["mut_aa"]]
287            for idx, mut in enumerate(af_missense_dict2[p_name2])
288            if str(res2) == split_missense_mutation(mut)[1]
289            and af_missense_dict2[p_name2][mut]["v_pathogenicity"] == "Likely pathogenic"
290        ]
291        afm_patho_mutations_2[res2] = f"({", ".join(afm_patho_mutations_2_)})"
292
293    mutated_res_1_new = {
294        k: ", ".join(v) if len(v) > 0 else "" for k, v in mutated_res_1.items()
295    }
296    mutated_res_2_new = {
297        k: ", ".join(v) if len(v) > 0 else "" for k, v in mutated_res_2.items()
298    }
299
300    for k, v in mutated_res_1.items():
301        if v == "":
302            new_v = f"{afm_patho_mutations_1[k]}"
303        else:
304            new_v = mutated_res_1_new[k] + f" {afm_patho_mutations_1[k]}"
305        mutated_res_1_new[k] = new_v.strip()
306
307    for k, v in mutated_res_2.items():
308        if v == "":
309            new_v = f"{afm_patho_mutations_2[k]}"
310        else:
311            new_v = mutated_res_2_new[k] + f" {afm_patho_mutations_2[k]}"
312        mutated_res_2_new[k] = new_v.strip()
313
314    cocomap_df["Mutated 1"] = cocomap_df["Res. Number 1"].map(mutated_res_1_new)
315    cocomap_df["Mutated 2"] = cocomap_df["Res. Number 2"].map(mutated_res_2_new)
316
317    cocomap_df["AF Missense 1 Score"] = cocomap_df["Res. Number 1"].map(
318        lambda x: np.mean(afm_score_res_1[x])
319    )
320    cocomap_df["AF Missense 2 Score"] = cocomap_df["Res. Number 2"].map(
321        lambda x: np.mean(afm_score_res_2[x])
322    )
323
324    return cocomap_df

372    """ Add and save COCOMAP results with AlphaFold metrics (PAE, pLDDT) columns.
373
374    #TODO: Throw error if csv files not found in cocomap_output_dir
375
376    ## Arguments:
377
378    - **result_metadata (dict)**:<br />
379        Metadata dictionary containing information about the result, including
380        'proteins' key with list of protein names for chain1 & 2, and 'structure_path'
381        key with path to the structure file used for COCOMAP analysis.
382
383    - **cocomap_output_dir (str)**:<br />
384        Directory where COCOMAP results are stored.
385

387        Metadata dictionary containing information about the AlphaFold result, including:
388        - 'structure_path': Path to the AlphaFold-predicted structure file (PDB).
389        - 'data_path': Path to the AlphaFold output data file (pickle or npz) containing PAE and pLDDT values.
390        - 'af_offset' (optional): Dictionary specifying residue numbering offsets for each chain,

392
393    - **rep_atom_dict (dict, optional):**:<br />
394        Dictionary mapping residue names to representative atom names.
395        This is used to determine which atom's pLDDT to use for a given residue.
396        If not provided, the code will attempt to use standard representative atoms
397        (e.g., CB/CA for amino acids) or fall back to the first atom in the residue.
398
399    - **average_token_pae (bool, optional):**:<br />
400        If True, use the average PAE over all tokens corresponding to a residue for that residue's PAE value.
401
402    - **average_token_plddt (bool, optional):**:<br />
403        If True, use the average pLDDT over all tokens corresponding to a residue for that residue's pLDDT value.
404
405    - **metric_level (str, optional):**:<br />
406        Level at which to compute metrics: 'per_token' or 'representative_token'.
407
408    - **mark_clashes (bool, optional):**:<br />
409        If True, mark interactions that are clashes as defined by COCOMAPS2.
410        COCOMAPS2 annotates interaction types with "*" if the distance between
411        the interacting atoms is less than the sum of their Van der Waals radii.
412        Note: There is currently a bug where clashes are not being correctly marked
413        in the final_file.csv, so this may not work as expected.
414    """
415
416    try:
417        from af_pipeline.parser.initialize import Initialize
418
419    except ImportError:
420        raise ImportError(
421            "af_pipeline module is required for adding AF metrics. "
422        )
423
424    af_struct_path = af_metadata["structure_path"]

439

442        structure_file_path=af_struct_path,
443        af_offset=af_offset,
444        rep_atom_dict=rep_atom_dict,
445        average_token_pae=average_token_pae,
446        average_token_plddt=average_token_plddt,
447        metric_level=metric_level,
448    )
449

451

453
454        final_csvs = [f for f in csv_files if f.endswith("final_file.csv")]

456            f"""Multiple final CSVs found for chain pair {chain_1},{chain_2}"""

461            print("-"*100)

463                csv_f.split(".")[2]
464                .replace(f"{struct_type}_{chain_1}_{chain_2}_", "")
465                .replace(f"{struct_type}_{chain_2}_{chain_1}_", "")
466            )

479            df_res_2 = list(df["Res. Number 2"])
480            df_res_names_1 = list(df["Res. Name 1"])
481            df_res_names_2 = list(df["Res. Name 2"])
482

485

506            atom2_plddts = af3_metrics["atom2_plddts"]

508            atom2_names = af3_metrics["atom2_names"]
509            metric_atom1_names = af3_metrics["metric_atom1_names"]
510            metric_atom2_names = af3_metrics["metric_atom2_names"]
511            pae_values = af3_metrics["pae_values"]
512            contact_prob_values = af3_metrics["contact_prob_values"]
513
514            assert len(pae_values) == len(atom1_plddts) == len(atom2_plddts), (
515                "Length mismatch in computed AF metrics."
516            )
517
518            df["AF PAE"] = pae_values
519            df["Atom 1 pLDDT"] = atom1_plddts
520            df["Atom 2 pLDDT"] = atom2_plddts
521
522            if len(contact_prob_values) == len(pae_values):
523                df["AF Contact Probability"] = contact_prob_values
524
525            new_csv_path = os.path.join(
526                cocomap_output_dir,
527                "af_metrics",
528                csv_f.replace(".csv", "_with_AF_metrics.csv")
529            )
530            os.makedirs(os.path.dirname(new_csv_path), exist_ok=True)
531            df.to_csv(new_csv_path, index=False, encoding='utf-8')
532
533            df_dict = {
534                "Res. Name 1": df_res_names_1,
535                "Res. Number 1": df_res_1,
536                chains_set_1_prot: df_chains_1,
537                "Res. Name 2": df_res_names_2,
538                "Res. Number 2": df_res_2,
539                chains_set_2_prot: df_chains_2,
540                "Type of Interactions": df_interaction_types,
541                "AF PAE": pae_values,
542                "Atom 1 pLDDT": atom1_plddts,
543                "Atom 2 pLDDT": atom2_plddts,
544                "Atom 1": atom1_names,
545                "Atom 2": atom2_names,
546                "Metric Atom 1": metric_atom1_names,
547                "Metric Atom 2": metric_atom2_names,
548            }
549
550            if len(contact_prob_values) == len(pae_values):
551                df_dict["AF Contact Probability"] = contact_prob_values
552
553            df_to_concat = pd.DataFrame(df_dict)
554            master_df_list.append(df_to_concat)
555
556        af_final_csv = pd.concat(master_df_list, ignore_index=True)
557        af_final_csv_path = os.path.join(
558            cocomap_output_dir,
559            "af_metrics",
560            final_csvs[0].replace(".csv", "_with_AF_metrics.csv")
561        )
562        os.makedirs(os.path.dirname(af_final_csv_path), exist_ok=True)
563        af_final_csv.to_excel(af_final_csv_path.replace(".csv", ".xlsx"), index=False)

574    - **rep_atom_dict (dict)**:<br />
575        Dictionary mapping residue names to representative atom names. Used to determine
576        which atom's pLDDT to use for a given residue.
577

579        Initialized AlphaFold parser object containing the structure and data needed to
580        extract metrics.
581
582    - **zipped_cra (zip)**:<br />
583        Zipped chain, residue, atom information for interacting pairs.
584
585    ## Returns:
586
587    - **_type_**:<br />
588        Dictionary containing extracted AF3 metrics, including:
589    """
590
591    atom1_idxs = []
592    atom2_idxs = []
593    atom1_plddts = []
594    atom2_plddts = []
595    atom1_names = []
596    atom2_names = []
597    metric_atom1_names = []
598    metric_atom2_names = []
599
600    for (chain_1, res_1, atom_1, chain_2, res_2, atom_2) in zipped_cra:
601        atom1_names.append(atom_1)
602        atom2_names.append(atom_2)
603
604        atom1_idx = get_atom_idx(
605            num_to_idx=initializer.num_to_idx,
606            chain=chain_1,
607            res=res_1,
608            atom=atom_1,
609        )
610
611        atom2_idx = get_atom_idx(
612            num_to_idx=initializer.num_to_idx,
613            chain=chain_2,
614            res=res_2,
615            atom=atom_2,
616        )
617
618        atom1_idxs.append(atom1_idx)
619        atom2_idxs.append(atom2_idx)
620
621        atom1_plddt, metric_atom1 = get_atom_plddt(
622            rep_atom_dict,
623            initializer,
624            chain_1,
625            res_1,
626            atom_1
627        )
628
629        atom2_plddt, metric_atom2 = get_atom_plddt(
630            rep_atom_dict,
631            initializer,
632            chain_2,
633            res_2,
634            atom_2
635        )
636
637        metric_atom1_names.append(metric_atom1)
638        metric_atom2_names.append(metric_atom2)
639
640        atom1_plddts.append(atom1_plddt)
641        atom2_plddts.append(atom2_plddt)
642
643    pae_values = []
644    for atoms1, atoms2 in zip(atom1_idxs, atom2_idxs):
645        pae_submatrix = initializer.avg_pae[atoms1, :][:, atoms2]
646        pae_values.append(pae_submatrix.mean().item())
647
648    contact_prob_values = []
649    if initializer.contact_probs is not None:
650        for atoms1, atoms2 in zip(atom1_idxs, atom2_idxs):
651            _ij = initializer.contact_probs[atoms1, :][:, atoms2]
652            _ji = initializer.contact_probs[atoms2, :][:, atoms1]
653            contact_prob_value = (_ij.mean().item() + _ji.mean().item()) / 2
654            contact_prob_values.append(contact_prob_value)
655
656    af3_metrics = {
657        "atom1_plddts": atom1_plddts,
658        "atom2_plddts": atom2_plddts,
659        "atom1_names": atom1_names,
660        "atom2_names": atom2_names,
661        "metric_atom1_names": metric_atom1_names,
662        "metric_atom2_names": metric_atom2_names,
663        "pae_values": pae_values,
664        "contact_prob_values": contact_prob_values,
665    }
666
667    return af3_metrics
