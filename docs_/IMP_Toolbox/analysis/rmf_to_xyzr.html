 69    def convert_rmf_to_xyzr(self) -> dict:
 70        """ Wrapper over `batch_worker` to parallely process frame batches and get
 71        the XYZR data for all specified frames.
 72
 73        ## Returns:
 74
 75        - **dict**:<br />
 76            Dictionary of moleculewise XYZR data for all processed frames.
 77            - key: molecule name with copy index  and residue/range
 78                (e.g. "mol1_0_1-10" or "mol1_0_11")
 79            - value: list of [x, y, z, r] for each bead in the molecule across
 80                all processed frames.
 81        """
 82
 83        molwise_xyzr = defaultdict(list)
 84
 85        with ProcessPoolExecutor(max_workers=self.num_cores) as executor:
 86            futures = [
 87                executor.submit(RMFToXYZRConverter.batch_worker, self.rmf_file, batch)
 88                for batch in self.frame_batches
 89            ]
 90            results = []
 91            for future in tqdm.tqdm(as_completed(futures), total=len(futures)):
 92                results.append(future.result())
 93
 94        for res in results:
 95            for bead_name, xyzr_list in res.items():
 96                molwise_xyzr[bead_name].extend(xyzr_list)
 97
 98        molwise_xyzr = dict(molwise_xyzr)
 99
100        return molwise_xyzr

102    def prepare_frame_batches(self) -> list:
103        """ Prepare required frame batches.
104
105        Frame batches are list of arrays of frame indices.
106        Each array will be processed by a separate worker in parallel.
107
108        ## Returns:
109
110        - **list**:<br />
111            List of arrays of frame indices for each batch.
112        """
113
114        frame_indices = np.arange(self.num_frames)
115
116        if self.frame_subset is not None:
117            frame_indices = get_res_range_from_key(self.frame_subset)
118            frame_indices = sorted(set(frame_indices))
119            frame_indices = np.array([i for i in frame_indices if i < self.num_frames])
120            self.num_frames = len(frame_indices)
121
122        frame_batches = np.array_split(np.arange(self.num_frames), self.num_cores)
123        frame_batches = [np.array(batch) for batch in frame_batches if len(batch) > 0]
124        frame_batches = [frame_indices[batch] for batch in frame_batches]
125
126        return frame_batches

128    @staticmethod
129    def batch_worker(
130        rmf_file: str,
131        frame_batch: np.ndarray,
132    ) -> dict:
133        """ Worker function to process a batch of frames from the RMF file.
134
135        - For each frame, it extracts the XYZR data for each molecule
136        and organizes it in a dictionary.
137        - key: molecule name with copy index (e.g. "mol1_0")
138        - value: dictionary of fragments with their XYZR data
139        (e.g. {"1-10": [[x, y, z, r], ...], "11": [[x, y, z, r], ...]})
140
141        ## Arguments:
142
143        - **rmf_path (str)**:<br />
144            Path to the RMF file.
145
146        - **frame_batch (np.ndarray)**:<br />
147            Array of frame indices to process in this batch.
148
149        ## Returns:
150
151        - **dict**:<br />
152            Dictionary of moleculewise XYZR data for the processed frames.
153            - key: molecule name with copy index and residue/range
154                (e.g. "mol1_0_1-10" or "mol1_0_11")
155            - value: list of [x, y, z, r] for each bead in the molecule across
156                all processed frames.
157        """
158
159        mol_rmf_dict = defaultdict(list)
160        mdl = IMP.Model()
161        rmf_data = RMF.open_rmf_file_read_only(rmf_file)
162        hier = IMP.rmf.create_hierarchies(rmf_data, mdl)[0]
163
164        for frame_id in tqdm.tqdm(frame_batch, smoothing=0):
165
166            IMP.rmf.load_frame(rmf_data, RMF.FrameID(frame_id))
167            mdl.update()
168            sel = IMP.atom.Selection(hier, resolution=1)
169
170            for leaf in sel.get_selected_particles():
171
172                p = IMP.core.XYZR(leaf)
173                coord = list(p.get_coordinates())
174                radius = p.get_radius()
175
176                bead_name = RMFToXYZRConverter.get_bead_name(leaf)
177                mol_rmf_dict[bead_name].append(coord + [radius])
178
179        mol_rmf_dict = dict(mol_rmf_dict)
180
181        del rmf_data
182        return mol_rmf_dict

469    def merge_residue_selection_by_copies(self) -> dict:
470        """ Merge moleculwise residue dictionary across molecule copies.
471
472        For example, if there are two copies of a molecule "MOL1_0" and "MOL1_1",
473        and "MOL1_0" has residues [1, 2, 3] and "MOL1_1" has residues [4, 5, 6],
474        after merging, both "MOL1_0" and "MOL1_1" will have residues
475        [1, 2, 3, 4, 5, 6].
476        """
477
478        merged_molwise_residues = defaultdict(list)
479
480        for mol in self.molwise_residues.keys():
481
482            base_mol = self.get_mol_name(mol, only_base=True)
483            merged_molwise_residues[base_mol].extend(self.molwise_residues[mol])
484
485        self.molwise_residues = {
486            k: sorted(set(v)) for k, v in merged_molwise_residues.items()
487        }

528    def filter_xyzr_data(
529        self,
530        sel_molwise_residues: dict,
531    ) -> dict:
532        """ Update xyzr_data to keep only beads corresponding to sel_molwise_residues.
533
534        ## Returns:
535
536        - **dict**:<br />
537            Updated xyzr_data dictionary with only the relevant beads.
538        """
539
540        keys_to_del = []
541        keys_to_update = {}
542
543        for bead_k, _xyzr in self.xyzr_data.items():
544
545            mol, sel = bead_k.rsplit("_", 1)
546            res_nums = set(get_res_range_from_key(sel))
547            req_res_nums = sel_molwise_residues.get(mol, set())
548
549            if len(req_res_nums) == 0:
550                keys_to_del.append(bead_k)
551                continue
552
553            elif len(res_nums.intersection(req_res_nums)) == 0:
554                keys_to_del.append(bead_k)
555                continue
556
557            intersecting_res = sorted(res_nums.intersection(req_res_nums))
558            n_bead_k = f"{mol}_{get_key_from_res_range(intersecting_res)}"
559            if n_bead_k == bead_k:
560                continue
561
562            keys_to_update[bead_k] = n_bead_k
563
564        for k in keys_to_del:
565            del self.xyzr_data[k]
566
567        for old_k, new_k in keys_to_update.items():
568            self.xyzr_data[new_k] = self.xyzr_data[old_k]
569            del self.xyzr_data[old_k]
570
571        return self.xyzr_data

609    @staticmethod
610    def get_mol_name(m: str, only_base: bool = False) -> str:
611        """ Extract molecule name with copy index from the bead key.
612
613        ## Arguments:
614
615        - **m (str)**:<br />
616            Molecule name in the expected format.
617            See `EXPECTED_MOLNAME_FORMATS` in `analysis_constants.py` for the
618            expected formats.
619
620        - **only_base (bool, optional):**:<br />
621            If True, only the base molecule name without copy index will be returned.
622
623        ## Returns:
624
625        - **str**:<br />
626            Molecule name with copy index if only_base is False, else only the base
627            molecule name.
628        """
629
630        _match = re.match(REGEX_MOLNAME, m)
631        if _match is None or len(_match.groups()) != 3:
632            raise ValueError(
633                f"Invalid format for molecule name: {m}. "\
634                f"Expected format is one of: {', '.join(EXPECTED_MOLNAME_FORMATS)}"
635            )
636        base, cp_idx, _ = _match.groups()
637
638        if only_base or cp_idx is None:
639            return base
640
641        return f"{base}{MOL_COPY_SEP}{cp_idx}"
