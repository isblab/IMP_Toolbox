136    def generate_chimerax_cmds_for_mapped_mutations(
137        self,
138        outpath: str,
139        mutated_regions: list = ["interface", "core", "exposed", "ptm"],
140        color: bool = True,
141    ):
142        """ Generate a .cxc file with ChimeraX commands to select and optionally
143        color residues based on their mapped mutation regions.
144
145        ## Arguments:
146
147        - **outpath (str)**:<br />
148            Path to save the generated .cxc file.
149
150        - **mutated_regions (list, optional):**:<br />
151            List of mutated regions to include in the ChimeraX commands.
152
153        - **color (bool, optional):**:<br />
154            Whether to color the selected residues based on their mutated regions.
155        """
156
157        self.check_if_mapped()
158        chimerax_cmds = []
159        for mutated_region, selections in self.chimerax_res_sels.items():
160            if mutated_region not in mutated_regions:
161                continue
162            for chain, res_nums in selections.items():
163                res_range = get_key_from_res_range(res_nums)
164                chimerax_cmds.append(f"sel add $1/{chain}:{res_range}")
165                if color:
166                    chimerax_cmds.append(
167                        f"col $1/{chain}:{res_range} {RES_COLOR_MAP[mutated_region]}"
168                    )
169
170        if not outpath.endswith(".cxc"):
171            outpath += ".cxc"
172
173        with open(outpath, "w") as f:
174            for cmd in chimerax_cmds:
175                f.write(cmd + "\n")
176
177        print("""
178        Use the following commands in ChimeraX to execute the generated commands:
179        open <your_ccm_file>. OR open <your_pdb_file>.pdb
180        runscript <your_generated_commands>.cxc #<model_id>
181        """)

183    def generate_chimerax_attributes_for_mapped_mutations(
184        self,
185        outpath: str,
186        operation: str = "average",
187        match_mode: str = "any",
188    ):
189
190        self.check_if_mapped()
191        chimerax_attrs = []
192        for sel_key, scores in self.afm_attr_scores.items():
193            if operation == "average":
194                avg_score = sum(scores) / len(scores)
195            elif operation == "max":
196                avg_score = max(scores)
197            elif operation == "min":
198                avg_score = min(scores)
199            afm_attr = f"\t{sel_key}\t{avg_score}"
200            chimerax_attrs.append(afm_attr)
201
202        attribute_template = textwrap.dedent(f"""
203        attribute: afm_score
204        match mode: {match_mode}
205        recipient: residues
206        """)
207        chimerax_attrs = attribute_template + "\n".join(chimerax_attrs)
208
209        if not outpath.endswith(".defattr"):
210            outpath += ".defattr"
211
212        with open(outpath, "w") as f:
213            f.write(chimerax_attrs)
214
215        print("""
216        Use the following commands in ChimeraX to execute the generated attribute definitions:
217        open <your_ccm_file>.rmf3 OR open <your_pdb_file>.pdb
218        open <your_generated_attributes>.defattr format defattr models #<model_id>
219        """)

254    def get_mutations_mapped_to_pdb(self):
255        """ Obtain mutations mapped to the given pdb.
256
257        ## Returns:
258
259        - **tuple**:<br />
260            A tuple containing two dictionaries:
261            1. A dictionary where keys are mutated regions and values are dictionaries
262                with chains as keys and sets of residue numbers as values.
263            2. A dictionary where keys are residue selections and values are lists of
264                AlphaMissense scores corresponding to those selections.
265        """
266
267        self.get_mapped_residues()
268
269        for idx, row in self.mutation_impact_df.iterrows():
270            protein = row["Protein"]
271            if len(self.modeled_ranges[protein]) == 2:
272                start, end = self.modeled_ranges[protein]
273            else:
274                start, end = 0, 0
275            gene = row["Gene"]
276            mutations = row["Mutation"].split("\n")
277            afm_scores = row["AlphaMissense Score"].split("\n")
278
279            for mutation, afm_score in zip(mutations, afm_scores):
280
281                wt, res_num, mut = split_missense_mutation(mutation)
282                chains = self.molwise_chain_map.get(protein, [])
283
284                if (
285                    int(res_num) not in range(start, end+1) or
286                    float(afm_score) < float(self.af_missense_cutoff)
287                ):
288                    continue
289
290                for chain in chains:
291
292                    if int(res_num) in self.mapped_interface.get(gene, set()):
293                        self.chimerax_res_sels["interface"][chain].add(res_num)
294                    elif int(res_num) in self.mapped_core.get(gene, set()):
295                        self.chimerax_res_sels["core"][chain].add(res_num)
296                    elif int(res_num) in self.mapped_exposed.get(gene, set()):
297                        self.chimerax_res_sels["exposed"][chain].add(res_num)
298                    if int(res_num) in self.mapped_ptm.get(gene, set()):
299                        self.chimerax_res_sels["ptm"][chain].add(res_num)
300
301                    self.afm_attr_scores[f"/{chain}:{res_num}"].append(afm_score)
302
303        return self.chimerax_res_sels, self.afm_attr_scores

408    def define_centroids(
409        self,
410        outpath: str,
411        mutated_regions: list = ["interface", "core", "exposed", "ptm"]
412    ):
413
414        reverse_chain_map = {v: k for k, v in self.chain_map.items()}
415
416        centroid_sel_cmd = Template("define centroid ${selection} radius 2.0 color ${color}")
417        centroids_cmds = []
418        for mutated_region, selections in self.chimerax_res_sels.items():
419            if mutated_region not in mutated_regions:
420                continue
421            for chain, res_nums in selections.items():
422                protein = reverse_chain_map[chain]
423                for res_num in res_nums:
424                    if "," in self.resolution_map[protein][res_num]:
425                        continue
426                    centroid_cmd = centroid_sel_cmd.substitute(
427                        selection=f"$1/{chain}:{res_num}",
428                        color=RES_COLOR_MAP[mutated_region]
429                    )
430                    centroids_cmds.append(centroid_cmd)
431
432        centroids_cmds.append(f"sel $1.1-{len(centroids_cmds)}")
433        centroids_cmds = "\n".join(centroids_cmds)
434
435        if not outpath.endswith(".cxc"):
436            outpath += ".cxc"
437
438        with open(outpath, "w") as f:
439            f.write(centroids_cmds)
