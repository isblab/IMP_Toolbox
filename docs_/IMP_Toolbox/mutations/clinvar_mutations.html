 122def get_variant_details_from_clinvar(
 123    variant_ids:list,
 124    api_parameters: dict = {},
 125    ignore_error: bool = False,
 126    max_retries: int = 3,
 127) -> dict:
 128    """ Fetch variant details from ClinVar for a list of variant IDs.
 129
 130    We get info in XML format from the API, which we convert and store as JSON.
 131
 132    > [!CAUTION]
 133    > If you parallelize it, make sure to not exceed API's rate limits.
 134    > Currently, we batch the requests in batches of 100 variant IDs. i.e.
 135    > if there are 1000 variant IDs, one call to `get_variant_details_from_clinvar`
 136    > will make 10 API calls, each with 100 variant IDs.
 137
 138    ## Arguments:
 139
 140    - **variant_ids (list)**:<br />
 141        List of ClinVar variant IDs.
 142
 143    - **api_parameters (dict)**:<br />
 144        API parameters for the request.
 145
 146    - **ignore_error (bool, optional)**:<br />
 147        Whether to ignore errors. Defaults to False.
 148
 149    - **max_retries (int, optional)**:<br />
 150        Maximum number of retries for the request. Defaults to 3.
 151
 152    ## Returns:
 153
 154    - **dict**:<br />
 155        Dictionary with variant IDs as keys and their details as values.
 156    """
 157
 158    variant_id_batches = [
 159        variant_ids[i:i + 100] for i in range(0, len(variant_ids), 100)
 160    ]
 161
 162    variant_info_dict = {}
 163
 164    for idx, batch in enumerate(tqdm(variant_id_batches)):
 165
 166        api_parameters["id"] = ",".join(batch)
 167
 168        req_sess = request_session(max_retries=max_retries)
 169        response = req_sess.get(API_URLS["ncbi_efetch"], params=api_parameters)
 170
 171        if response.status_code == 200:
 172            result = response.text
 173        else:
 174            if ignore_error:
 175                return {}
 176            else:
 177                raise ValueError(f"Error fetching variant details for {batch}")
 178
 179        root = ET.fromstring(result)
 180        variation_info = root.findall("VariationArchive")
 181        variant_ids = [
 182            var_id_elem.get("VariationID") for var_id_elem in variation_info
 183        ]
 184
 185        for idx, variation_id in enumerate(variant_ids):
 186            variant_info_dict[variation_id] = xmltodict.parse(
 187                ET.tostring(variation_info[idx])
 188            )
 189
 190    return variant_info_dict

 192def fetch_clinvar_variant_data(
 193    save_path: str,
 194    input_data: str | list,
 195    api_parameters: dict = {},
 196    ignore_error: bool = False,
 197    max_retries: int = 3,
 198    overwrite: bool = False,
 199) -> dict | list:
 200    """ Fetch ClinVar variant data for given gene name or variant IDs.
 201
 202    Uses `get_variant_ids_from_clinvar` or `get_variant_details_from_clinvar`.
 203
 204    ## Arguments:
 205
 206    - **save_path (str)**:<br />
 207        Path to save the fetched variant data.
 208
 209    - **input_data (str | list)**:<br />
 210        Gene name (str) or list of variant IDs (list).
 211
 212    - **api_parameters (dict, optional)**:<br />
 213        API parameters for the request. Defaults to {}.
 214
 215    - **ignore_error (bool, optional)**:<br />
 216        Whether to ignore errors. Defaults to False.
 217
 218    - **max_retries (int, optional)**:<br />
 219        Maximum number of retries for the request. Defaults to 3.
 220
 221    - **overwrite (bool, optional)**:<br />
 222        Whether to overwrite existing data at save_path. Defaults to False.
 223
 224    ## Returns:
 225
 226    - **dict | list**:<br />
 227        Fetched variant data.
 228    """
 229
 230    data_type_dict = {str: "ids", list: "details"}
 231
 232    try:
 233        data_type = data_type_dict[type(input_data)]
 234
 235    except KeyError:
 236        raise ValueError(
 237            f"Invalid input data type: {type(input_data)}. "
 238            "Expected str (gene name) or list (variant ids)."
 239        )
 240
 241    if os.path.exists(save_path) and overwrite is False:
 242        variant_data = read_json(save_path)
 243        return variant_data
 244
 245    func_getter_ = {
 246        "ids": get_variant_ids_from_clinvar,
 247        "details": get_variant_details_from_clinvar,
 248    }
 249
 250    get_func = func_getter_.get(data_type)
 251
 252    assert callable(get_func), f"{get_func} is not callable."
 253
 254    variant_data = get_func(
 255        input_data,
 256        api_parameters=api_parameters,
 257        ignore_error=ignore_error,
 258        max_retries=max_retries,
 259    )
 260
 261    write_json(save_path, variant_data)
 262
 263    return variant_data

 396

 432    def get_molecular_consequence_list(
 433        self,
 434        missense_only: bool = True,
 435    ) -> list:
 436        """ Get the molecular consequence list from the HGVS list for a given
 437        RefSeq ID.
 438
 439        At least one molecular consequence should be "missense variant".
 440
 441        ## Arguments:
 442
 443        - **missense_only (bool, optional)**:<br />
 444            Whether to filter for missense variants only. Defaults to True.
 445
 446        ## Returns:
 447
 448        - **list**:<br />
 449            List of molecular consequences.
 450        """
 451
 452        def check_mc_type(mc: str) -> bool:
 453            if missense_only:
 454                return mc.get("@Type", "") == "missense variant"
 455            return True
 456
 457        molecular_consequence_list = []
 458
 459        if isinstance(self.hgvs_list.get("HGVS", []), dict):
 460            self.hgvs_list["HGVS"] = [self.hgvs_list["HGVS"]]
 461
 462        for hgvs_entry in self.hgvs_list.get("HGVS", []):
 463
 464            # we only want consequence at protein level
 465            if hgvs_entry.get("@Type", "") != "coding":
 466                continue
 467
 468            # depending on the sequence, molecular consequence can differ
 469            # we only want those that match the RefSeq ID in the title
 470            # check if the sequence identifier matches
 471            nucleotide_express = hgvs_entry.get("NucleotideExpression", {})
 472            seq_accession = nucleotide_express.get("@sequenceAccession", "")
 473
 474            # BUG: potential bug
 475            # can cause issues if the aa sequence changed in the newer
 476            # versions of same base RefSeq ID
 477            # "@sequenceAccessionVersion" provides the full ID, but it can be
 478            # different from the one in the title
 479            if (
 480                self.ncbi_ref_seq_id is None
 481                or seq_accession != self.ncbi_ref_seq_id.split(".")[0]
 482            ):
 483                continue
 484
 485            mol_consequence = hgvs_entry.get("MolecularConsequence", {})
 486
 487            if isinstance(mol_consequence, list):
 488                for mc in mol_consequence:
 489                    if check_mc_type(mc):
 490                        molecular_consequence_list.extend([
 491                            mc.get("@Type", "") for mc in mol_consequence
 492                        ])
 493                        break
 494
 495            elif isinstance(mol_consequence, dict):
 496                if check_mc_type(mol_consequence):
 497                    molecular_consequence_list.append(
 498                        mol_consequence.get("@Type", "")
 499                    )
 500
 501        return list(set(molecular_consequence_list))

 752    def get_all_clinical_assertions(
 753        self,
 754        sort_by_date: bool=True,
 755        date_format:str=DATE_FORMAT,
 756    ) -> list:
 757        """ Get all clinical assertions from a classified record.
 758
 759        Clinical assertion represents a single submission to ClinVar corresponding
 760        to the variant. Multiple submissions can be present for a single variant.
 761        We only look at those that contribute to the agg. clinical significance.
 762
 763        if `sort_by_date` is True, the assertions are sorted by the date they
 764        were last evaluated. If the evaluation date is not available, the date
 765        last updated is used instead.
 766
 767        Look through all the assertions that contributed to aggregate clinical
 768        significance
 769
 770        ## Arguments:
 771
 772        - **sort_by_date (bool, optional)**:<br />
 773            Whether to sort the assertions by date. Defaults to True.
 774
 775        - **date_format (str, optional)**:<br />
 776            Date format for parsing dates. Defaults to DATE_FORMAT.
 777
 778        ## Returns:
 779
 780        - **list**:<br />
 781            List of clinical assertions.
 782        """
 783
 784        clinical_assertions = []
 785
 786        for clinical_assertion in self.clinical_assertion_list:
 787            # we look at only those that contribute
 788            contributes_to_agg_class = clinical_assertion.get(
 789                "@ContributesToAggregateClassification", "false"
 790            )
 791            if contributes_to_agg_class == "false":
 792                continue
 793
 794            last_updated = clinical_assertion.get("@DateLastUpdated", "")
 795            assertion_class = clinical_assertion.get("Classification", {})
 796
 797            germline_class = assertion_class.get("GermlineClassification", "")
 798            last_evaluated = assertion_class.get("@DateLastEvaluated", "")
 799            assertion_comment = assertion_class.get("Comment", "")
 800
 801            if isinstance(assertion_comment, dict):
 802                assertion_comment = assertion_comment.get("#text", "")
 803
 804            last_updated = datetime.strptime(last_updated, date_format)
 805            try:
 806                last_evaluated = datetime.strptime(last_evaluated, date_format)
 807            except ValueError:
 808                last_evaluated = last_updated
 809
 810            clinical_assertions.append({
 811                "last_updated_date": last_updated,
 812                "asserted_germline_class": germline_class,
 813                "assertion_comment": assertion_comment,
 814                "last_evaluated_date": last_evaluated,
 815            })
 816
 817        if sort_by_date:
 818            clinical_assertions = sorted(
 819                clinical_assertions,
 820                key=lambda x: x["last_evaluated_date"],
 821                reverse=True
 822            )
 823
 824        return clinical_assertions

 877    def is_invalid_variant(self) -> bool:
 878        """ Check if the variant is invalid based on certain conditions.
 879
 880        The conditions checked are:
 881        - Aggregate clinical significance is not in allowed list.
 882        - NCBI RefSeq ID is None.
 883        - Molecular consequence is not valid (i.e. not a missense variant).
 884        - Provided gene name doesn't match NCBI gene name (weird but true, DSC2)
 885        - Mutation is not a single missense mutation.
 886        - If agg. significance is "Conflicting classifications of pathogenicity",
 887            the most recent significance is not in allowed list.
 888
 889        ## Returns:
 890
 891        - **bool**:<br />
 892            True if the variant is invalid, False otherwise.
 893        """
 894
 895        condition_val = (
 896            self.agg_significance not in CLINVAR_ALLOWED_CLINICAL_SIGNIFICANCE
 897            or self.ncbi_ref_seq_id is None
 898            or len(self.molecular_consequence_list) == 0
 899            or self.g_name != self.ncbi_g_name
 900            or is_missense_mutation(self.p_mutation, ignore_warnings=True) is False
 901            or (self.agg_significance
 902                == "Conflicting classifications of pathogenicity"
 903                and self.all_significances[0]
 904                not in CLINVAR_ALLOWED_CLINICAL_SIGNIFICANCE)
 905        )
 906
 907        return condition_val

 909    def update_p_mutation(
 910        self,
 911        modeled_seq: str,
 912        ref_seq_file: str,
 913        pairwise_alignment_file: str,
 914        ignore_warnings: bool = True,
 915    ):
 916        """ Update the protein mutation based on the modeled sequence.
 917
 918        The ClinVar protein mutation is based on the NCBI reference sequence.
 919        The modeled sequence might differ due to different isoforms, this
 920        function updates the residue number in the protein mutation accordingly.
 921
 922        ## Arguments:
 923
 924        - **modeled_seq (str)**:<br />
 925            Sequence of the modeled protein.
 926
 927        - **ref_seq_file (str)**:<br />
 928            Path to the NCBI reference sequence JSON file.
 929
 930        - **pairwise_alignment_file (str)**:<br />
 931            Path to the pairwise alignment file.
 932
 933        - **ignore_warnings (bool, optional)**:<br />
 934            Whether to ignore warnings. Defaults to True.
 935
 936        ## Returns:
 937
 938        - **str | None**:<br />
 939            Updated protein mutation or None if p_mutation is None.
 940        """
 941
 942        if self.p_mutation is None or self.ncbi_ref_seq_id is None:
 943            return None
 944
 945        p_sequence = get_ncbi_ref_seq(self.ncbi_ref_seq_id, ref_seq_file)
 946
 947        pairwise_seq_alignment = PairwiseSequenceAlignment(
 948            seq1=modeled_seq,
 949            seq2=p_sequence,
 950        )
 951        clinvar_psa_map = pairwise_seq_alignment.fetch_pairwise_alingment_map(
 952            pairwise_alignment_file=pairwise_alignment_file,
 953            overwrite=False,
 954        )
 955
 956        # warn if sequences not identical
 957        if p_sequence != modeled_seq and ignore_warnings is False:
 958            warnings.warn(
 959                f"""Sequences not identical for {self.p_name}.
 960                Cross-check the pairwise alignment.\n
 961                Pairwise alignment at {pairwise_alignment_file}."""
 962            )
 963
 964        split_mut = split_missense_mutation(
 965            p_mutation=self.p_mutation,
 966            return_type="all",
 967            ignore_warnings=True,
 968        )
 969
 970        if split_mut is None:
 971            self.p_mutation = None
 972            return None
 973
 974        wt_aa, res_num, mut_aa = split_mut
 975        res_num = int(res_num)
 976
 977        res_num_mapped, warn_msg = PairwiseSequenceAlignment.get_mapped_residue(
 978            psa_map=clinvar_psa_map,
 979            codon_number=res_num,
 980        )
 981
 982        if ignore_warnings is False and len(warn_msg) > 0:
 983            warnings.warn(f"""{warn_msg} (Protein: {self.p_name})""")
 984
 985        if len(clinvar_psa_map) == 0:
 986            res_num_mapped = res_num
 987        else:
 988            try:
 989                res_num_mapped = clinvar_psa_map[res_num]
 990            except KeyError:
 991                warnings.warn(
 992                    f"""
 993                    Residue number {res_num} not found in
 994                    pairwise alignment map for protein {self.p_name}.
 995                    Skipping...
 996                    """
 997                ) if ignore_warnings is False else None
 998                res_num_mapped = None
 999
1000        self.p_mutation = (
1001            None if res_num_mapped is None
1002            else f"{wt_aa}{res_num_mapped}{mut_aa}"
1003        )
1004
1005        return self.p_mutation

 911        self,
 912        modeled_seq: str,
 913        ref_seq_file: str,
 914        pairwise_alignment_file: str,
 915        ignore_warnings: bool = True,
 916    ):
 917        """ Update the protein mutation based on the modeled sequence.
 918
 919        The ClinVar protein mutation is based on the NCBI reference sequence.
 920        The modeled sequence might differ due to different isoforms, this
 921        function updates the residue number in the protein mutation accordingly.
 922
 923        ## Arguments:
 924
 925        - **modeled_seq (str)**:<br />
 926            Sequence of the modeled protein.
 927
 928        - **ref_seq_file (str)**:<br />

 930
 931        - **pairwise_alignment_file (str)**:<br />
 932            Path to the pairwise alignment file.
 933
 934        - **ignore_warnings (bool, optional)**:<br />
 935            Whether to ignore warnings. Defaults to True.
 936
 937        ## Returns:
 938
 939        - **str | None**:<br />
 940            Updated protein mutation or None if p_mutation is None.
 941        """
 942
 943        if self.p_mutation is None or self.ncbi_ref_seq_id is None:
 944            return None
 945
 946        p_sequence = get_ncbi_ref_seq(self.ncbi_ref_seq_id, ref_seq_file)
 947
 948        pairwise_seq_alignment = PairwiseSequenceAlignment(
 949            seq1=modeled_seq,
 950            seq2=p_sequence,
 951        )
 952        clinvar_psa_map = pairwise_seq_alignment.fetch_pairwise_alingment_map(
 953            pairwise_alignment_file=pairwise_alignment_file,
 954            overwrite=False,
 955        )
 956
 957        # warn if sequences not identical
 958        if p_sequence != modeled_seq and ignore_warnings is False:
 959            warnings.warn(
 960                f"""Sequences not identical for {self.p_name}.
 961                Cross-check the pairwise alignment.\n
 962                Pairwise alignment at {pairwise_alignment_file}."""
 963            )
 964
 965        split_mut = split_missense_mutation(
 966            p_mutation=self.p_mutation,
 967            return_type="all",
 968            ignore_warnings=True,
 969        )
 970
 971        if split_mut is None:
 972            self.p_mutation = None
 973            return None
 974
 975        wt_aa, res_num, mut_aa = split_mut
 976        res_num = int(res_num)
 977

 979            psa_map=clinvar_psa_map,
 980            codon_number=res_num,
 981        )

 985
 986        if len(clinvar_psa_map) == 0:
 987            res_num_mapped = res_num

 993                    f"""
